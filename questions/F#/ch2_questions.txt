let addTen numbers =

// Возвести числа в квадрат и оставить только чётные квадраты
let squareAll numbers = Seq.map (fun n -> n * n) numbers
let keepEven numbers = Seq.filter (fun n -> n % 2 = 0) numbers
let evenSquares numbers = (squareAll >> keepEven) numbers

evenSquares ?    // ? — список целых чисел (int list), вход теста

Требование: функция evenSquares получает список целых чисел (по модулю не больше 50 000) и возвращает последовательность (seq) квадратов чётных чисел списка в исходном порядке. Квадрат нечётного числа нечётен, поэтому нечётные числа в результат не попадают. Каждый элемент результата должен быть точным значением n × n, то есть неотрицательным числом.
