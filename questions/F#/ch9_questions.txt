#r "nuget:FSToolkit.ErrorHandling, 2.13.0"
open FsToolkit.ErrorHandling
open System

// Разбирает строку в целое число
let tryParseNumber (text:string) =
    match Int32.TryParse text with
    | true, number -> Some number
    | false, _ -> None

// Сумма заказа: цена × количество − скидка
let orderTotal (priceText:string) (qtyText:string) (discountText:string) =
    option {
        let! price = tryParseNumber priceText
        let! qty = tryParseNumber qtyText
        let! discount = tryParseNumber discountText
        return price * qty - discount
    }

Справка: Int32.TryParse в F# возвращает кортеж (tuple) bool * int. Если строка — целое число без пробелов (например, "42" или "-5"), результат (true, 42) / (true, -5); иначе (например, "abc", "", "2.5") — (false, 0). В тестах используйте строки без пробелов. Строки #r и open подключают пакет FsToolkit.ErrorHandling с выражением option { } (computation expression).

Требование (requirement): функция orderTotal получает три строки из формы заказа — цену, количество и скидку — и возвращает int option:
- если цена и количество — целые числа, результат Some (цена × количество − скидка);
- скидка необязательна: пустая строка "" означает «скидки нет» (скидка 0);
- если цена, количество или непустая скидка не являются целым числом — None.
В тестах цена и количество не отрицательные, а скидка не больше, чем цена × количество.

Входные данные теста — три строки, например orderTotal "500" "3" "100".
