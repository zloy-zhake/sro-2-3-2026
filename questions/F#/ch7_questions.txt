// голы команд: collect → groupBy → map → maxBy и дальше
type Game =
    { Home: string; HomeGoals: int
      Away: string; AwayGoals: int }

let games =
    [ { Home = "Aktobe"; HomeGoals = 2; Away = "Ordabasy"; AwayGoals = 1 }
      { Home = "Ordabasy"; HomeGoals = 4; Away = "Zhetysu"; AwayGoals = 0 }
      { Home = "Zhetysu"; HomeGoals = 2; Away = "Aktobe"; AwayGoals = 2 } ]

let rows =
    games
    |> List.collect (fun g -> [ (g.Home, g.HomeGoals); (g.Away, g.AwayGoals) ])
let groups = rows |> List.groupBy (fun (team, _) -> team)
let goalsByTeam =
    groups
    |> List.map (fun (team, items) -> team, items |> List.sumBy (fun (_, n) -> n))
let top = goalsByTeam |> List.maxBy (fun (_, total) -> total)
let goalList = goalsByTeam |> List.map (fun (_, total) -> total)
let ordered = goalList |> List.sort
let pairs = ordered |> List.windowed 2
let diffs = pairs |> List.map (fun pair -> pair[1] - pair[0])

// Проверка инструктажа по технике безопасности
let briefingReport team briefed =
    let missing = team - briefed
    let missingCount = Set.count missing
    if Set.isEmpty team then "Команда пуста"
    elif team |> Set.isSubset briefed then "Все прошли инструктаж"
    else $"Не прошли: {missingCount}"

Справка:
- Set.isSubset a b возвращает true, если каждый элемент множества a есть во множестве b (a — подмножество (subset) b). Например, Set.isSubset (Set [ 1 ]) (Set [ 1; 2 ]) = true.
- a - b — разность множеств (Set.difference): элементы a, которых нет в b.
- Set [ ... ] создаёт множество (set); повторы удаляются.

Требование (requirement): функция briefingReport получает два множества имён: team — сотрудники команды, briefed — все, кто прошёл инструктаж в этот день (там могут быть и сотрудники других команд). Функция возвращает строку:

// Обрабатывает команды платёжного терминала до команды выхода
let runCommands commands =
    commands
    |> Seq.takeWhile (fun cmd -> cmd <> 'q')
    |> Seq.map (fun cmd ->
        if cmd = 'p' then "Оплата"
        elif cmd = 'r' then "Возврат"
        else $"Неизвестная команда {cmd}")
    |> Seq.toList

Справка: Seq.takeWhile условие — берёт элементы с начала последовательности, пока условие истинно, и останавливается на первом элементе, для которого оно ложно (этот элемент и все следующие отбрасываются). Список (list) можно передать туда, где ожидается последовательность (seq).

Требование (requirement): функция runCommands получает символы, введённые кассиром на терминале, и возвращает список сообщений (string list) — по одному на каждую обработанную команду, в порядке ввода:
- 'p' — "Оплата";
- 'r' — "Возврат";
- любой другой символ — "Неизвестная команда X", где X — этот символ.
Команда выхода — 'q'. На экране терминала написано «Q — выход», поэтому выходом считаются и 'q', и 'Q'. Команда выхода и всё, что введено после неё, не обрабатываются. Если команд нет — пустой список [].
Регистр важен только для выхода: 'P' и 'R' — неизвестные команды.

type Order = { Courier : string; Minutes : int }

type CourierStat = { Name : string; mutable OnTime : int }
let stats = ResizeArray<CourierStat>()

// Находит курьера, который чаще всех доставлял вовремя (не дольше 30 минут)
let bestCourier (orders : Order list) =
    for order in orders do
        if order.Minutes <= 30 then
            let mutable found = false
            for entry in stats do
                if entry.Name = order.Courier then
                    found <- true
                    entry.OnTime <- entry.OnTime + 1
            if not found then
                stats.Add { Name = order.Courier; OnTime = 1 }
    let mutable best = stats[0]
    for row in stats do
        if row.OnTime > best.OnTime then best <- row
    best.Name, best.OnTime

Справка: обращение по индексу к несуществующему элементу ResizeArray (например, stats[0], когда в stats нет ни одного элемента) вызывает исключение (exception) во время выполнения.

Требование (requirement): функцию bestCourier вызывают в конце каждого дня со списком заказов ЭТОГО дня. Каждый вызов должен считать только переданные ему заказы, независимо от предыдущих вызовов. Функция возвращает кортеж (tuple) (имя курьера, число заказов, доставленных вовремя):
- заказ доставлен вовремя, если Minutes не больше 30 (30 минут — ещё вовремя);
- побеждает курьер с наибольшим числом заказов, доставленных вовремя; при равенстве — тот, кто раньше встречается в списке среди заказов, доставленных вовремя;
- если вовремя не доставлено ни одного заказа (в том числе если список пуст) — ("нет", 0).

Входные данные теста — один или несколько вызовов подряд. Считайте, что перед КАЖДЫМ тестом код загружен заново (новая сессия FSI). Для краткости заказ можно записывать как (Aru, 20) вместо { Courier = "Aru"; Minutes = 20 }. Пример: bestCourier [ (Aru, 20); (Bek, 35) ].
