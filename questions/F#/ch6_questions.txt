Прочитайте код.

let subtract a b = a - b
let multiply a b = a * b

// ручная передача промежуточных значений
let step1 = subtract 4 20
let step2 = multiply 3 step1

// конвейер (pipeline)
let piped = 20 |> subtract 4 |> multiply 3

// функция с параметром и композиция (composition)
let calc input = input |> subtract 4 |> multiply 3
let calcShort = subtract 4 >> multiply 3

1. Значение piped равно 48: конвейер (pipeline) передаёт 20 первым аргументом в subtract, получается 20 - 4 = 16, затем 16 умножается на 3. (5 баллов)
2. Выражение multiply 3 (subtract 4 20) даёт то же значение, что piped и step2. (5 баллов)
3. В calcShort оператор композиции (composition) >> применяет функции справа налево: сначала multiply 3, затем subtract 4. (5 баллов)
4. Если дописать строку let bad = multiply 3 subtract 4 20, она скомпилируется и даст то же значение, что piped, потому что скобки в F# необязательны. (5 баллов)
5. Тип subtract — int -> int -> int: это функция одного аргумента, которая возвращает функцию (каррирование, currying). (4 балла)

module Trip =
    type TripResult =
        {
            FuelLeft : float
            NeedsRefuel : bool
        }

    // Поездка: остаток топлива и нужна ли заправка
    let travel distance fuel =
        let left =
            if distance > 60 then fuel - 12.0
            elif distance > 0 then fuel - 4.0
            else fuel
        {
            FuelLeft = left
            NeedsRefuel = left < 6.0
        }

open Trip

module Scores =
    let add x y = x + y
    let subtract x y = x - y

    module Penalty =
        let late = 15

        // Итоговый балл: сумма двух частей минус штраф за опоздание
        let finalScore first second isLate =
            let total =
                if isLate then add first second |> subtract late
                else add first second
            if total < 0 then 0 else total

// первые два ? — баллы за две части (int, от 0 до 50); третий ? — опоздал ли студент (bool)
Scores.Penalty.finalScore ? ? ?

Требование: итоговый балл равен сумме баллов за две части. Если студент опоздал (isLate = true), из суммы вычитается штраф 15 баллов. Итоговый балл не может быть меньше 0: если после штрафа получилось отрицательное число, результат — 0.
