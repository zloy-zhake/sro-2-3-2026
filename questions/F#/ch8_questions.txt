type Applicant =
    { Years : int
      HasOverdraft : bool
      Overdraft : int }

// Лимит овердрафта: 200 за каждый год истории
let maxOverdraft years = years * 200

// Можно ли выдать кредит клиенту
let canGetCredit applicant =
    match applicant with
    | { Years = 0 } -> false
    | { Years = 1; HasOverdraft = true } when applicant.Overdraft >= maxOverdraft 1 -> false
    | { Years = 2; HasOverdraft = true } when applicant.Overdraft >= maxOverdraft 2 -> false
    | _ -> true

Требование (requirement): функция canGetCredit получает клиента (запись Applicant: Years — полных лет кредитной истории, 0 или больше; HasOverdraft — есть ли у клиента овердрафт; Overdraft — сумма овердрафта) и возвращает true (кредит можно выдать) или false (нельзя):
- история 0 лет — false;
- история 1 или 2 года и есть овердрафт: лимит овердрафта — 200 за каждый год истории (1 год — 200, 2 года — 400); овердрафт НЕ БОЛЬШЕ лимита (в том числе ровно равный лимиту) — true, больше лимита — false;
- история 1 или 2 года и овердрафта нет (HasOverdraft = false) — true, поле Overdraft не учитывается;
- история 3 года и больше — true при любом овердрафте.

Входные данные теста — запись Applicant, например canGetCredit { Years = 1; HasOverdraft = true; Overdraft = 150 }.

type PhoneNumber =
    | Domestic of number:string
    | Abroad of countryCode:string * number:string

type Channel =
    | Mail of address:string
    | Call of PhoneNumber
    | Text of PhoneNumber

// Описывает, как будет доставлено уведомление
let notify (message:string) channel =
    match channel with
    | Mail address -> $"Письмо на {address}: {message}"
    | Call (Domestic number) -> $"Звонок на {number}"
    | Call (Abroad (code, number)) -> $"Звонок на {code} {number}"
    | Text (Domestic number) -> $"SMS на {number}: {message}"

Требование (requirement): функция notify получает текст уведомления и канал связи (Channel) и возвращает строку — как будет доставлено уведомление:
- Mail — "Письмо на <адрес>: <текст>";
- Call с местным номером (Domestic) — "Звонок на <номер>";
- Call с зарубежным номером (Abroad) — "Звонок на <код страны> <номер>";
- Text с местным номером — "SMS на <номер>: <текст>";
- Text с зарубежным номером — "SMS на <код страны> <номер>: <текст>".
Для любого канала функция должна вернуть строку.

Входные данные теста — два аргумента, например notify "Код 1234" (Mail "aru@mail.kz"). Ожидаемый результат — строка целиком.
